import pandas as pd

df = pd.read_excel("techgear_sales_data.xlsx")

# Sort by date to ensure time order
df = df.sort_values("Date")

# -----------------------------
# 3-Month Moving Average Forecast
# -----------------------------
moving_avg_forecast = df["Sales"].tail(3).mean()

# -----------------------------
# Exponential Smoothing Forecast (alpha = 0.80)
# -----------------------------
alpha = 0.80
sales = df["Sales"]

exp_forecast = sales.iloc[0]  # initialize with first value
for value in sales:
    exp_forecast = alpha * value + (1 - alpha) * exp_forecast

print("3-Month Moving Average Forecast:", round(moving_avg_forecast, 2))
print("Exponential Smoothing Forecast:", round(exp_forecast, 2))

3-Month Moving Average Forecast: 5444.33
Exponential Smoothing Forecast: 5362.68

import pandas as pd
from sklearn.model_selection import KFold, cross_val_score
from sklearn.linear_model import LinearRegression
from sklearn.tree import DecisionTreeRegressor
from sklearn.metrics import mean_squared_error
import numpy as np

# Load data
df = pd.read_excel("techgear_sales_data.xlsx")

X = df[["Ad_Spend_Facebook", "Ad_Spend_Instagram", "Discount_Rate"]]
y = df["Sales"]

# Models
lin_model = LinearRegression()
tree_model = DecisionTreeRegressor(random_state=0)

# RMSE via 5-fold cross validation
kf = KFold(n_splits=5, shuffle=True, random_state=1)

lin_rmse = np.sqrt(-cross_val_score(lin_model, X, y, scoring='neg_mean_squared_error', cv=kf)).mean()
tree_rmse = np.sqrt(-cross_val_score(tree_model, X, y, scoring='neg_mean_squared_error', cv=kf)).mean()

print("Linear Regression RMSE:", round(lin_rmse, 2))
print("Decision Tree RMSE:", round(tree_rmse, 2))

Linear Regression RMSE: 631.75
Decision Tree RMSE: 938.05

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

df = pd.read_excel("techgear_sales_data.xlsx")
sales = df["Sales"]

# Monte Carlo simulation (1,000 runs)
simulations = []
for i in range(1000):
    simulated_month = np.random.uniform(sales.min(), sales.max(), 30).mean()
    simulations.append(simulated_month)

simulations = np.array(simulations)

print("Average simulated monthly sales:", round(simulations.mean(), 2))
print("Median simulated monthly sales:", round(np.median(simulations), 2))
print("Standard deviation:", round(simulations.std(), 2))

# Visualization
plt.hist(simulations, bins=20)
plt.title("Distribution of Simulated Monthly Sales")
plt.xlabel("Simulated Monthly Sales")
plt.ylabel("Frequency")
plt.show()

# Linear Programming via simple search (no external libraries)

best_sales = -1
best_F = None
best_I = None

for F in range(2000, 10001):   # Facebook must be at least $2,000
    I = 10000 - F              # Total budget is $10,000, so Instagram is the rest

    # Instagram constraints:
    if I < 1000 or I > 7000:   # between $1,000 and $7,000
        continue
    if I < 0.5 * F:            # Instagram must be at least 50% of Facebook spend
        continue

    # Objective: maximize sales = 5F + 3I
    sales = 5 * F + 3 * I

    if sales > best_sales:
        best_sales = sales
        best_F = F
        best_I = I

print("Optimal Facebook Spend:", best_F)
print("Optimal Instagram Spend:", best_I)
print("Maximum Achievable Sales:", best_sales)

Optimal Instagram Spend: 3334
Maximum Achievable Sales: 43332
